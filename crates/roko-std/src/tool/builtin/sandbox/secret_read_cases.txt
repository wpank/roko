# Commands that roko-std's bash tool (sandbox.rs) and the Claude CLI guard
# (roko-agent's claude_cli_guard.py) must refuse (deny) or run (allow). Both
# test this table in one workspace: roko.toml holds a secret, roko.lock and
# src/a.rs sit beside it, the key files .roko/.env, .roko/secrets.toml and
# vendor/pkg/.roko/credentials.json are there too, and ".." is the directory
# above. "in src:" runs the command from src. Without the secret and the key
# files, every command runs.

# Recursive searches of a tree that holds the config.
deny grep -r api_key .
deny grep -R x
deny grep -rn x .
deny grep -ril x
deny grep --recursive x .
deny grep -d recurse x .
deny grep --directories=recurse x .
deny grep -r -e x .
deny grep -rnm1 x .
deny grep -re x
deny rgrep x
deny rg api_key
deny rg x .
deny rg -i x
deny rg -e x
deny rg -uu x
deny sudo grep -r x .
deny timeout 5 rg x
deny sh -c 'grep -r x .'
deny echo $(grep -r x .)
deny grep -r x src/..
deny grep -r --include='*.toml' x .
deny rg -g '*.toml' x
deny rg -g '*.{rs,toml}' x
deny rg --glob=**/*.toml x
deny rg -t toml x
deny git grep api_key
deny git grep -n key
deny git -C . grep key
deny git grep key -- '*.toml'
deny git grep key HEAD
deny git grep key roko.toml
deny sudo git grep key
deny ag api_key
deny ag key .
deny timeout 5 ag key
deny ack api_key
deny ack -i key

# Globs and braces that name the config.
deny cat *
deny grep x *
deny grep -r x *
deny head -5 roko.*
deny cat roko.{toml,lock}
deny head ro{ko,xx}.toml
deny grep key {roko.toml,src/a.rs}
deny grep -r --exclude=roko.toml x .

# Reads of a list the check cannot see.
deny find . -exec cat {} +
deny find . -type f -exec head -n 5 {} \;
deny find . -name '*.toml' -exec cat {} +
deny find . -name roko.toml -exec grep key {} +
deny find . -type f | xargs cat
deny find . | xargs grep key
deny find . -type f -print0 | xargs -0 tail
deny cat $(find . -name '*.toml')
deny fd -x cat
deny fd toml -x cat
deny ls | xargs cat
deny xargs cat < list.txt

# Recursive reads that reach a key file, with roko.toml left out.
deny grep -r --exclude=*.toml OPENAI .
deny grep -r --exclude='roko.*' OPENAI .
deny grep -r --include='*.env' OPENAI .
deny grep -r --include=.env OPENAI .
deny grep -r --exclude-dir=vendor --exclude='*.toml' OPENAI .
deny grep -r OPENAI vendor
deny grep -r OPENAI .roko
deny rgrep --exclude='*.toml' OPENAI
deny rg --hidden -g '!*.toml' OPENAI
deny rg -uu -g '!*.toml' OPENAI
deny rg -. -t sh OPENAI
deny rg -g '*' -g '!*.toml' OPENAI
deny rg OPENAI .roko
deny rg -t json -u -u OPENAI vendor
deny ag --hidden -G env OPENAI
deny ag -u --ignore='*.toml' OPENAI
deny ack --ignore-file=ext:toml OPENAI
deny ack OPENAI vendor
deny git grep --untracked OPENAI -- '*.env'
deny git grep --no-index OPENAI vendor
deny git -C vendor grep --untracked OPENAI
deny find . -name .env -exec cat {} +
deny find . -name 'secrets*' -exec cat {} +
deny find vendor -type f | xargs cat
deny find vendor -type f -not -name '*.rs' -exec cat {} +
deny find . -name x -o -exec cat {} +
deny find . -path ./src -prune -o -type f -exec cat {} +
deny find . \( -name '*.rs' -o -name '.e*' \) -exec cat {} +
deny fd -H -e env -x cat
deny fd -u secrets -x cat
deny fd -H . vendor -x cat
deny cd .roko && ls | xargs cat
deny in src: grep -r --exclude='*.toml' OPENAI ..
deny cd src && grep -r --exclude='*.toml' OPENAI ..

# A command after cd runs where cd leads.
deny cd src && grep -r key ..
deny cd src; rg key ..
deny in src: grep -r x ..
deny in src: rg x ..
deny in src: cd .. && rg key
deny in src: git grep key ..
deny in src: ag key ..

# Searches and reads that leave the config out.
allow grep -r x src
allow grep x src/a.rs
allow grep x
allow grep -r --include='*.rs' x .
allow grep -rl --include=*.md x .
allow grep -r --exclude='roko.*' --exclude-dir='.*' x .
allow grep -r --exclude=*.toml --exclude-dir='.*' x .
allow rg x src
allow rg -g '*.rs' x
allow rg -g '!roko.toml' x
allow rg -g '!*.toml' x
allow rg -t rust x
allow rg --type=py x
allow rg -T toml x
allow rg --files
allow git grep key -- '*.rs'
allow git grep key src
allow git grep -l key src/a.rs
allow git log --grep x
allow ag -G '\.rs$' key
allow ag -g toml
allow ack -t rust key
allow ack -f
allow cat src/a.rs
allow cat src/{a.rs,b.rs}
allow echo {a,b}
allow echo 'grep -r x .'
allow find . -name '*.rs' -exec cat {} +
allow find src -type f -exec cat {} +
allow find . -name '*.rs' | xargs cat
allow find src | xargs cat
allow find . -type f | wc -l
allow fd -e rs -x cat
allow ls | xargs wc -l
allow cd src && grep -r key .
allow cd src && rg key
allow cd src && cat a.rs
allow in src: grep -r x .
allow in src: rg x
allow in src: git grep key
allow grep -r --exclude='*.toml' --exclude-dir='.*' OPENAI .
allow grep -r --exclude-dir=.roko OPENAI vendor
allow rgrep --exclude='roko.*' --exclude-dir='.*' OPENAI
allow rg -g '.env' -g '!*.toml' OPENAI
allow rg --hidden -g '!.*' -g '!*.toml' OPENAI
allow rg --hidden -g '!.roko' OPENAI vendor
allow rg --hidden -t rust OPENAI
allow rg -uu --max-depth 1 -g '!*.toml' OPENAI
allow rg OPENAI vendor
allow ag OPENAI vendor
allow ag --hidden --ignore=.roko OPENAI vendor
allow ack --ignore-dir=.roko OPENAI vendor
allow ack --ignore-dir=is:.roko OPENAI vendor
allow git grep OPENAI -- '*.env'
allow git grep OPENAI vendor
allow git grep --untracked OPENAI -- '*.rs'
allow find vendor -name .roko -prune -o -type f -print | xargs cat
allow find vendor -maxdepth 2 -type f -exec cat {} +
allow find vendor -type f -not -path '*/.roko*' | xargs cat
allow find . \( -name '*.rs' -o -name '*.md' \) -exec cat {} +
allow find vendor -not -regex '.*/\.[^/]*/.*' -exec cat {} +
allow fd -e env -x cat
allow fd . vendor -x cat
allow fd -H -E .roko -e json . vendor -x cat
allow cd vendor && ls | xargs cat
allow cd vendor/pkg && ls -a | xargs cat
